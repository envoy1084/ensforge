# useDeleteSubname

Hook for deleting subname.

## Import

```tsx
import { useDeleteSubname } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useDeleteSubname } from "@ensforge/react";

function Component() {
  const mutation = useDeleteSubname();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
