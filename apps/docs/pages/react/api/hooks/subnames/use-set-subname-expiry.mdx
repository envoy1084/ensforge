# useSetSubnameExpiry

Hook for setting subname expiry.

## Import

```tsx
import { useSetSubnameExpiry } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetSubnameExpiry } from "@ensforge/react";

function Component() {
  const mutation = useSetSubnameExpiry();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          expiry: 2_000_000_000n,
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
