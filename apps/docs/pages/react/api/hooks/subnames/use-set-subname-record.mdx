# useSetSubnameRecord

Hook for setting subname record.

## Import

```tsx
import { useSetSubnameRecord } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetSubnameRecord } from "@ensforge/react";

function Component() {
  const mutation = useSetSubnameRecord();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          owner: "0x0000000000000000000000000000000000000001",
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
