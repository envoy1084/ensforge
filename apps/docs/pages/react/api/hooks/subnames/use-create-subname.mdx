# useCreateSubname

Hook for creating subname.

## Import

```tsx
import { useCreateSubname } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useCreateSubname } from "@ensforge/react";

function Component() {
  const mutation = useCreateSubname();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          owner: "0x0000000000000000000000000000000000000001",
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
