# useMigrateName

Hook for migrating name.

## Import

```tsx
import { useMigrateName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useMigrateName } from "@ensforge/react";

function Component() {
  const mutation = useMigrateName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
