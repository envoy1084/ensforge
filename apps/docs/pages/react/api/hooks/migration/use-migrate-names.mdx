# useMigrateNames

Hook for migrating names.

## Import

```tsx
import { useMigrateNames } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useMigrateNames } from "@ensforge/react";

function Component() {
  const mutation = useMigrateNames();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          migrations: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
