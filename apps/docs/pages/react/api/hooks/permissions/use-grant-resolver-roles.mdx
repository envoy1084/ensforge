# useGrantResolverRoles

Hook for granting resolver roles.

## Import

```tsx
import { useGrantResolverRoles } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useGrantResolverRoles } from "@ensforge/react";

function Component() {
  const mutation = useGrantResolverRoles();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          account: "value",
          record: {},
        })
      }
    >
      Submit
    </button>
  );
}
```
