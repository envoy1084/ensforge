# useRevokeResolverRoles

Hook for revoking resolver roles.

## Import

```tsx
import { useRevokeResolverRoles } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useRevokeResolverRoles } from "@ensforge/react";

function Component() {
  const mutation = useRevokeResolverRoles();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          account: "value",
          record: {},
        })
      }
    >
      Submit
    </button>
  );
}
```
