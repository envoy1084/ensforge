# useApproveName

Hook for approving name.

## Import

```tsx
import { useApproveName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useApproveName } from "@ensforge/react";

function Component() {
  const mutation = useApproveName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          approved: true,
        })
      }
    >
      Submit
    </button>
  );
}
```
