# useGrantRegistryRoles

Hook for granting registry roles.

## Import

```tsx
import { useGrantRegistryRoles } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useGrantRegistryRoles } from "@ensforge/react";

function Component() {
  const mutation = useGrantRegistryRoles();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          account: "value",
          roles: 1n,
        })
      }
    >
      Submit
    </button>
  );
}
```
