# useSetTtl

Hook for setting ttl.

## Import

```tsx
import { useSetTtl } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetTtl } from "@ensforge/react";

function Component() {
  const mutation = useSetTtl();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          ttl: 300n,
        })
      }
    >
      Submit
    </button>
  );
}
```
