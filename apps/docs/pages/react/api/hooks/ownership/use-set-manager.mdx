# useSetManager

Hook for setting manager.

## Import

```tsx
import { useSetManager } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetManager } from "@ensforge/react";

function Component() {
  const mutation = useSetManager();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          manager: "0x0000000000000000000000000000000000000001",
        })
      }
    >
      Submit
    </button>
  );
}
```
