# useSetAddresses

Hook for setting addresses.

## Import

```tsx
import { useSetAddresses } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetAddresses } from "@ensforge/react";

function Component() {
  const mutation = useSetAddresses();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          addresses: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
