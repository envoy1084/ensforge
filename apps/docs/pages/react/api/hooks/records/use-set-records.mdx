# useSetRecords

Hook for setting records.

## Import

```tsx
import { useSetRecords } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetRecords } from "@ensforge/react";

function Component() {
  const mutation = useSetRecords();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          records: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
