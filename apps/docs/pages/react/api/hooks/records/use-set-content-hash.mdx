# useSetContentHash

Hook for setting content hash.

## Import

```tsx
import { useSetContentHash } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetContentHash } from "@ensforge/react";

function Component() {
  const mutation = useSetContentHash();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          protocol: "ipfs",
          value: "https://example.com",
        })
      }
    >
      Submit
    </button>
  );
}
```
