# useSetTexts

Hook for setting texts.

## Import

```tsx
import { useSetTexts } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetTexts } from "@ensforge/react";

function Component() {
  const mutation = useSetTexts();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          texts: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
