# useSetAddress

Hook for setting address.

## Import

```tsx
import { useSetAddress } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetAddress } from "@ensforge/react";

function Component() {
  const mutation = useSetAddress();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          address: "0x0000000000000000000000000000000000000001",
        })
      }
    >
      Submit
    </button>
  );
}
```
