# useSetAlias

Hook for setting alias.

## Import

```tsx
import { useSetAlias } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetAlias } from "@ensforge/react";

function Component() {
  const mutation = useSetAlias();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          target: "0x0000000000000000000000000000000000000001",
        })
      }
    >
      Submit
    </button>
  );
}
```
