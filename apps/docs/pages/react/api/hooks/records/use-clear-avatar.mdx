# useClearAvatar

Hook for clearing avatar.

## Import

```tsx
import { useClearAvatar } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useClearAvatar } from "@ensforge/react";

function Component() {
  const mutation = useClearAvatar();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
