# useSetData

Hook for setting data.

## Import

```tsx
import { useSetData } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetData } from "@ensforge/react";

function Component() {
  const mutation = useSetData();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          key: "url",
          value: "https://example.com",
        })
      }
    >
      Submit
    </button>
  );
}
```
