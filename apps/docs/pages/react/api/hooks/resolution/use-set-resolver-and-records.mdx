# useSetResolverAndRecords

Hook for setting resolver and records.

## Import

```tsx
import { useSetResolverAndRecords } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetResolverAndRecords } from "@ensforge/react";

function Component() {
  const mutation = useSetResolverAndRecords();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          records: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
