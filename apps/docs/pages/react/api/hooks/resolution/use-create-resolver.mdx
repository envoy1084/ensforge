# useCreateResolver

Hook for creating resolver.

## Import

```tsx
import { useCreateResolver } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useCreateResolver } from "@ensforge/react";

function Component() {
  const mutation = useCreateResolver();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          salt: 1n,
        })
      }
    >
      Submit
    </button>
  );
}
```
