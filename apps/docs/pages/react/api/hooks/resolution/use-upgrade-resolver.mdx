# useUpgradeResolver

Hook for upgrading resolver.

## Import

```tsx
import { useUpgradeResolver } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useUpgradeResolver } from "@ensforge/react";

function Component() {
  const mutation = useUpgradeResolver();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
