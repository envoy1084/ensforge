# useSetResolver

Hook for setting resolver.

## Import

```tsx
import { useSetResolver } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetResolver } from "@ensforge/react";

function Component() {
  const mutation = useSetResolver();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          resolver: "0x0000000000000000000000000000000000000001",
        })
      }
    >
      Submit
    </button>
  );
}
```
