# useClaimDnsName

Hook for claiming dns name.

## Import

```tsx
import { useClaimDnsName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useClaimDnsName } from "@ensforge/react";

function Component() {
  const mutation = useClaimDnsName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          proof: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
