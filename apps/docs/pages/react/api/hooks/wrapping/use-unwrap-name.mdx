# useUnwrapName

Hook for unwrapping name.

## Import

```tsx
import { useUnwrapName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useUnwrapName } from "@ensforge/react";

function Component() {
  const mutation = useUnwrapName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          manager: "0x0000000000000000000000000000000000000001",
        })
      }
    >
      Submit
    </button>
  );
}
```
