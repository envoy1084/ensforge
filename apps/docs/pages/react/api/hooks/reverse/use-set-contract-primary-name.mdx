# useSetContractPrimaryName

Hook for setting contract primary name.

## Import

```tsx
import { useSetContractPrimaryName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetContractPrimaryName } from "@ensforge/react";

function Component() {
  const mutation = useSetContractPrimaryName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          contract: "value",
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
