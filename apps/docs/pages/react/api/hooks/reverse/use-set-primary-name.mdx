# useSetPrimaryName

Hook for setting primary name.

## Import

```tsx
import { useSetPrimaryName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSetPrimaryName } from "@ensforge/react";

function Component() {
  const mutation = useSetPrimaryName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
        })
      }
    >
      Submit
    </button>
  );
}
```
