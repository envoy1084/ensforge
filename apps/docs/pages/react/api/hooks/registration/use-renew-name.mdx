# useRenewName

Hook for renewing name.

## Import

```tsx
import { useRenewName } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useRenewName } from "@ensforge/react";

function Component() {
  const mutation = useRenewName();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          name: "example.eth",
          duration: 365n * 24n * 60n * 60n,
        })
      }
    >
      Submit
    </button>
  );
}
```
