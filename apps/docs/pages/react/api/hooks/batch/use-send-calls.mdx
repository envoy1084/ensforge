# useSendCalls

Hook for sending calls.

## Import

```tsx
import { useSendCalls } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useSendCalls } from "@ensforge/react";

function Component() {
  const mutation = useSendCalls();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          calls: [],
        })
      }
    >
      Submit
    </button>
  );
}
```
