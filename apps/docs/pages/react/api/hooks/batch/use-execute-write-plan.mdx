# useExecuteWritePlan

Hook for executing write plan.

## Import

```tsx
import { useExecuteWritePlan } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useExecuteWritePlan } from "@ensforge/react";

function Component() {
  const mutation = useExecuteWritePlan();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          plan: { id: "profile", stages: [] },
        })
      }
    >
      Submit
    </button>
  );
}
```
