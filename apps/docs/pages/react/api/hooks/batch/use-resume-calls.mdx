# useResumeCalls

Hook for resuming calls.

## Import

```tsx
import { useResumeCalls } from "@ensforge/react";
```

## Usage

:::code-group

```tsx [component.tsx]
import { useResumeCalls } from "@ensforge/react";

function Component() {
  const mutation = useResumeCalls();

  return (
    <button
      disabled={mutation.isWaiting}
      onClick={() =>
        mutation.mutate({
          batch: {},
        })
      }
    >
      Submit
    </button>
  );
}
```
